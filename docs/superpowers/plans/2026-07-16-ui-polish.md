```kotlin
package com.esusic.recoverytracker.model

import kotlin.math.abs

enum class DeltaKind { BETTER, WORSE, NEUTRAL }

data class MetricDelta(val diff: Double, val kind: DeltaKind)

fun metricDelta(
    current: Double?,
    baseline: Double?,
    higherIsBetter: Boolean?,
    neutralBandFraction: Double = 0.03
): MetricDelta? {
    if (current == null || baseline == null) return null
    val diff = current - baseline
    val kind = when {
        higherIsBetter == null -> DeltaKind.NEUTRAL
        abs(diff) <= abs(baseline) * neutralBandFraction -> DeltaKind.NEUTRAL
        (diff > 0) == higherIsBetter -> DeltaKind.BETTER
        else -> DeltaKind.WORSE
    }
    return MetricDelta(diff, kind)
}
```

```kotlin
import androidx.compose.ui.text.font.FontWeight
```

```kotlin
import com.esusic.recoverytracker.model.MetricDelta
import com.esusic.recoverytracker.ui.theme.AppColors
import kotlin.math.abs

@Composable
fun DeltaChip(
    delta: MetricDelta?,
    unit: String,
    modifier: Modifier = Modifier,
    decimals: Int = 0,
    suffix: String = "vs 30d"
) {
    if (delta == null) return
    val color = when (delta.kind) {
        DeltaKind.BETTER -> AppColors.Recovery
        DeltaKind.WORSE -> AppColors.Danger
        DeltaKind.NEUTRAL -> AppColors.TextSecondary
    }
    val arrow = when {
        delta.diff > 0 -> "▲"
        delta.diff < 0 -> "▼"
        else -> "—"
    }
    val magnitude = "%.${decimals}f".format(abs(delta.diff))
    val valueText = if (unit.isEmpty()) magnitude else "$magnitude $unit"
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(50))
            .background(color.copy(alpha = 0.12f))
            .padding(horizontal = 8.dp, vertical = 3.dp)
    ) {
        Text(
            text = "$arrow $valueText $suffix",
            color = color,
            fontSize = 11.sp,
            fontWeight = FontWeight.SemiBold
        )
    }
}
```

```kotlin
data class TodayUiState(
    val recoveryValue: Int? = null,
    val isCalibrating: Boolean = false,
    val strainValue: Double? = null,
    val sleepPerformance: Int? = null,
    val sleepDurationMinutes: Long? = null,
    val restingHeartRate: Long? = null,
    val avgHrvMillis: Double? = null,
    val steps: Int? = null,
    val avgRespiratoryRate: Double? = null,
    val avgOxygenSaturationPercent: Double? = null,
    val avgSkinTemperatureDeltaCelsius: Double? = null,
    val latestHeartRateBpm: Long? = null,
    val latestHeartRateEpochMillis: Long? = null,
    val lastSyncedAtEpochMillis: Long? = null,
    val hasAllPermissions: Boolean = true,
    val isRefreshing: Boolean = false,
    val hrvSubScore: Double? = null,
    val rhrSubScore: Double? = null,
    val sleepSubScore: Double? = null,
    val rhrDelta: MetricDelta? = null,
    val hrvDelta: MetricDelta? = null,
    val respRateDelta: MetricDelta? = null,
    val spo2Delta: MetricDelta? = null,
    val skinTempDelta: MetricDelta? = null,
    val sleepDelta: MetricDelta? = null
)

class TodayViewModel(
    private val repository: HealthRepository,
    private val healthDataSource: HealthDataSource
) : ViewModel() {

    private val _uiState = MutableStateFlow(TodayUiState())
    val uiState: StateFlow<TodayUiState> = _uiState.asStateFlow()

    init {
        refresh()
    }

    fun refresh() {
        viewModelScope.launch {
            _uiState.value = _uiState.value.copy(isRefreshing = true)

            val hasPermissions = healthDataSource.hasAllPermissions()
            if (!hasPermissions) {
                _uiState.value = _uiState.value.copy(hasAllPermissions = false, isRefreshing = false)
                return@launch
            }

            val score = repository.syncToday()
            val aggregates = repository.observeRecentAggregates(BaselineCalculator.WINDOW_DAYS + 1).first()
            val aggregate = aggregates.firstOrNull()
            val history = aggregates.drop(1)

            fun baselineMean(values: List<Double>): Double? =
                BaselineCalculator.calculate(values).takeIf { it.sampleCount > 0 }?.mean

            val rhrBaseline = baselineMean(history.mapNotNull { it.restingHeartRate?.toDouble() })
            val hrvBaseline = baselineMean(history.mapNotNull { it.avgHrvMillis })
            val respBaseline = baselineMean(history.mapNotNull { it.avgRespiratoryRate })
            val spo2Baseline = baselineMean(history.mapNotNull { it.avgOxygenSaturationPercent })
            val skinTempBaseline = baselineMean(history.mapNotNull { it.avgSkinTemperatureDeltaCelsius })
            val sleepBaseline = baselineMean(history.mapNotNull { it.sleepDurationMinutes?.toDouble() })

            _uiState.value = TodayUiState(
                recoveryValue = score.recoveryValue,
                isCalibrating = score.recoveryIsCalibrating,
                strainValue = score.strainValue,
                sleepPerformance = score.sleepPerformance,
                sleepDurationMinutes = aggregate?.sleepDurationMinutes,
                restingHeartRate = aggregate?.restingHeartRate,
                avgHrvMillis = aggregate?.avgHrvMillis,
                steps = aggregate?.steps,
                avgRespiratoryRate = aggregate?.avgRespiratoryRate,
                avgOxygenSaturationPercent = aggregate?.avgOxygenSaturationPercent,
                avgSkinTemperatureDeltaCelsius = aggregate?.avgSkinTemperatureDeltaCelsius,
                latestHeartRateBpm = aggregate?.latestHeartRateBpm,
                latestHeartRateEpochMillis = aggregate?.latestHeartRateEpochMillis,
                lastSyncedAtEpochMillis = score.lastSyncedAtEpochMillis,
                hasAllPermissions = true,
                isRefreshing = false,
                hrvSubScore = score.hrvSubScore,
                rhrSubScore = score.rhrSubScore,
                sleepSubScore = score.sleepSubScore,
                rhrDelta = metricDelta(aggregate?.restingHeartRate?.toDouble(), rhrBaseline, higherIsBetter = false),
                hrvDelta = metricDelta(aggregate?.avgHrvMillis, hrvBaseline, higherIsBetter = true),
                respRateDelta = metricDelta(aggregate?.avgRespiratoryRate, respBaseline, higherIsBetter = null),
                spo2Delta = metricDelta(aggregate?.avgOxygenSaturationPercent, spo2Baseline, higherIsBetter = true, neutralBandFraction = 0.01),
                skinTempDelta = metricDelta(aggregate?.avgSkinTemperatureDeltaCelsius, skinTempBaseline, higherIsBetter = null),
                sleepDelta = metricDelta(aggregate?.sleepDurationMinutes?.toDouble(), sleepBaseline, higherIsBetter = true, neutralBandFraction = 0.05)
            )
        }
    }
```

```kotlin
package com.esusic.recoverytracker.ui.trends

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.esusic.recoverytracker.data.HealthRepository
import com.esusic.recoverytracker.model.MetricDelta
import com.esusic.recoverytracker.model.metricDelta
import com.esusic.recoverytracker.ui.common.ChartPoint
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.flatMapLatest
import kotlinx.coroutines.launch
import java.time.LocalDate

enum class TrendRange(val days: Int) { WEEK(7), MONTH(30), QUARTER(90) }

data class MetricTrend(
    val points: List<ChartPoint> = emptyList(),
    val latest: Float? = null,
    val delta: MetricDelta? = null
)

data class TrendsUiState(
    val range: TrendRange = TrendRange.WEEK,
    val recovery: MetricTrend = MetricTrend(),
    val strain: MetricTrend = MetricTrend(),
    val hrv: MetricTrend = MetricTrend(),
    val rhr: MetricTrend = MetricTrend()
)

class TrendsViewModel(private val repository: HealthRepository) : ViewModel() {

    private val selectedRange = MutableStateFlow(TrendRange.WEEK)
    private val _uiState = MutableStateFlow(TrendsUiState())
    val uiState: StateFlow<TrendsUiState> = _uiState.asStateFlow()

    init {
        viewModelScope.launch {
            selectedRange.flatMapLatest { range ->
                combine(
                    repository.observeRecentScores(range.days),
                    repository.observeRecentAggregates(range.days)
                ) { scores, aggregates ->
                    val chronologicalScores = scores.reversed()
                    val chronologicalAggregates = aggregates.reversed()
                    TrendsUiState(
                        range = range,
                        recovery = buildTrend(
                            chronologicalScores.map { ChartPoint(LocalDate.parse(it.date), it.recoveryValue.toFloat()) },
                            higherIsBetter = true
                        ),
                        strain = buildTrend(
                            chronologicalScores.map { ChartPoint(LocalDate.parse(it.date), it.strainValue.toFloat()) },
                            higherIsBetter = null
                        ),
                        hrv = buildTrend(
                            chronologicalAggregates.mapNotNull { a -> a.avgHrvMillis?.let { ChartPoint(LocalDate.parse(a.date), it.toFloat()) } },
                            higherIsBetter = true
                        ),
                        rhr = buildTrend(
                            chronologicalAggregates.mapNotNull { a -> a.restingHeartRate?.let { ChartPoint(LocalDate.parse(a.date), it.toFloat()) } },
                            higherIsBetter = false
                        )
                    )
                }
            }.collect { _uiState.value = it }
        }
    }

    private fun buildTrend(points: List<ChartPoint>, higherIsBetter: Boolean?): MetricTrend {
        val latest = points.lastOrNull()?.value
        val priorAverage = points.dropLast(1)
            .map { it.value.toDouble() }
            .takeIf { it.isNotEmpty() }
            ?.average()
        return MetricTrend(
            points = points,
            latest = latest,
            delta = metricDelta(latest?.toDouble(), priorAverage, higherIsBetter)
        )
    }

    fun selectRange(range: TrendRange) {
        selectedRange.value = range
    }
}
```

```kotlin
import java.time.format.DateTimeFormatter

private val dateFormatter = DateTimeFormatter.ofPattern("MMM d", Locale.ENGLISH)

@Composable
fun MetricLineChart(
    points: List<ChartPoint>,
    lineColor: Color,
    modifier: Modifier = Modifier,
    valueFormatter: (Float) -> String = { "%.0f".format(it) },
    onSelect: (ChartPoint?) -> Unit = {}
) {
    val textMeasurer = rememberTextMeasurer()
    val labelStyle = TextStyle(fontSize = 10.sp, color = AppColors.TextMuted)
    var selectedIndex by remember(points) { mutableStateOf<Int?>(null) }

    Column(modifier = modifier, verticalArrangement = Arrangement.spacedBy(4.dp)) {
        Canvas(
            modifier = Modifier
                .fillMaxWidth()
                .height(150.dp)
                .pointerInput(points) {
                    detectDragGestures(
                        onDrag = { change, _ ->
                            val index = nearestPointIndex(points, change.position.x, size.width.toFloat())
                            selectedIndex = index
                            onSelect(index?.let { points[it] })
                        },
                        onDragEnd = {
                            selectedIndex = null
                            onSelect(null)
                        }
                    )
                }
        ) {
            if (points.size < 2) return@Canvas

            val minValue = points.minOf { it.value }
            val maxValue = points.maxOf { it.value }
            val range = (maxValue - minValue).takeIf { it > 0f } ?: 1f
            val verticalInset = 8.dp.toPx()
            val plotHeight = size.height - 2 * verticalInset
            val firstDay = points.first().date.toEpochDay()
            val daySpan = (points.last().date.toEpochDay() - firstDay).coerceAtLeast(1).toFloat()

            fun xFor(point: ChartPoint): Float =
                ((point.date.toEpochDay() - firstDay) / daySpan) * size.width

            fun yFor(value: Float): Float =
                verticalInset + (1f - (value - minValue) / range) * plotHeight

            listOf(maxValue, minValue + range / 2f, minValue).forEach { gridValue ->
                val y = yFor(gridValue)
                drawLine(color = AppColors.GridLine, start = Offset(0f, y), end = Offset(size.width, y), strokeWidth = 1f)
                val measured = textMeasurer.measure(AnnotatedString(valueFormatter(gridValue)), labelStyle)
                drawText(measured, topLeft = Offset(0f, y - measured.size.height - 2f))
            }

            val linePath = Path()
            points.forEachIndexed { index, point ->
                val offset = Offset(xFor(point), yFor(point.value))
                if (index == 0) linePath.moveTo(offset.x, offset.y) else linePath.lineTo(offset.x, offset.y)
            }

            val fillPath = Path().apply {
                addPath(linePath)
                lineTo(xFor(points.last()), size.height)
                lineTo(xFor(points.first()), size.height)
                close()
            }
            drawPath(
                path = fillPath,
                brush = Brush.verticalGradient(
                    colors = listOf(lineColor.copy(alpha = 0.22f), Color.Transparent),
                    startY = 0f,
                    endY = size.height
                )
            )
            drawPath(path = linePath, color = lineColor, style = Stroke(width = 2.dp.toPx(), cap = StrokeCap.Round))
```

```kotlin
                    style = MaterialTheme.typography.bodySmall,
                    color = AppColors.TextMuted
                )
            }
            return@Column
        }

            Text(
                "${total / 60}h ${total % 60}m",
                style = MaterialTheme.typography.displaySmall,
                fontWeight = FontWeight.Bold
            )
            DeltaChip(uiState.durationDelta, "min")
        }

        val sessionStart = uiState.sessionStart
        val sessionEnd = uiState.sessionEnd
        if (sessionStart != null && sessionEnd != null && uiState.heartRateBuckets.isNotEmpty()) {

                    buckets = uiState.heartRateBuckets,
                    sessionStart = sessionStart,
                    sessionEnd = sessionEnd
                )
            }
        }

                    stagePeriods = uiState.stagePeriods,

                    sessionEnd = sessionEnd

            StageLegendRow(SleepStageType.REM, uiState.remMinutes, uiState.remPercent)
            StageLegendRow(SleepStageType.LIGHT, uiState.lightMinutes, uiState.lightPercent)
        }
    }

                style = MaterialTheme.typography.bodyMedium,
                color = AppColors.TextSecondary,
                fontWeight = FontWeight.SemiBold
            )
        }
        Box(
            modifier = Modifier.fillMaxWidth().height(6.dp).clip(RoundedCornerShape(3.dp))
                .background(MaterialTheme.colorScheme.surfaceVariant)
        ) {
            Box(
                modifier = Modifier.fillMaxWidth(percent.coerceIn(0f, 1f)).fillMaxHeight()
                    .clip(RoundedCornerShape(3.dp)).background(color)
            )
        }
    }
}
```
