```kotlin
package com.esusic.recoverytracker.scoring

import kotlin.math.ln

data class VitalAgeInputs(
    val avgHrvMillis: Double? = null,
    val avgRestingHr: Double? = null,
    val avgSleepMinutes: Double? = null,
    val avgSteps: Double? = null,
    val avgStrain: Double? = null
)

data class VitalAgeComponent(val offsetYears: Double, val weight: Double)

data class VitalAgeResult(
    val vitalAge: Double,
    val chronologicalAge: Int,
    val hrv: VitalAgeComponent?,
    val rhr: VitalAgeComponent?,
    val sleep: VitalAgeComponent?,
    val activity: VitalAgeComponent?
)

object VitalAgeCalculator {

    private enum class Component(val baseWeight: Double) {
        HRV(0.40), RHR(0.25), SLEEP(0.20), ACTIVITY(0.15)
    }

    // Log-linear fit through Umetani et al. (1998) population RMSSD-by-age norms
    // (~45 ms in the mid-20s down to ~25 ms in the mid-60s).
    private const val HRV_LN_INTERCEPT = 4.18
    private const val HRV_DECAY_PER_YEAR = 0.0148

    fun calculate(chronologicalAge: Int, inputs: VitalAgeInputs): VitalAgeResult? {
        val offsets = buildMap {
            hrvOffset(inputs.avgHrvMillis, chronologicalAge)?.let { put(Component.HRV, it) }
            rhrOffset(inputs.avgRestingHr)?.let { put(Component.RHR, it) }
            sleepOffset(inputs.avgSleepMinutes)?.let { put(Component.SLEEP, it) }
            activityOffset(inputs.avgSteps, inputs.avgStrain)?.let { put(Component.ACTIVITY, it) }
        }
        if (offsets.size < 2) return null

        val totalWeight = offsets.keys.sumOf { it.baseWeight }
        val components = offsets.mapValues { (component, offset) ->
            VitalAgeComponent(offset, component.baseWeight / totalWeight)
        }
        val vitalAge = (chronologicalAge + components.values.sumOf { it.offsetYears * it.weight })
            .coerceAtLeast(18.0)

        return VitalAgeResult(
            vitalAge = vitalAge,
            chronologicalAge = chronologicalAge,
            hrv = components[Component.HRV],
            rhr = components[Component.RHR],
            sleep = components[Component.SLEEP],
            activity = components[Component.ACTIVITY]
        )
    }

    private fun hrvOffset(rmssdMillis: Double?, chronologicalAge: Int): Double? {
        if (rmssdMillis == null || rmssdMillis <= 0.0) return null
        val hrvAge = ((HRV_LN_INTERCEPT - ln(rmssdMillis)) / HRV_DECAY_PER_YEAR).coerceIn(18.0, 80.0)
        return (hrvAge - chronologicalAge).coerceIn(-10.0, 10.0)
    }

    private fun rhrOffset(restingHr: Double?): Double? =
        restingHr?.let { ((it - 62.0) * 0.4).coerceIn(-10.0, 10.0) }

    private fun sleepOffset(minutes: Double?): Double? = minutes?.let { m ->
        when {
            m in 420.0..510.0 -> -1.0
            m < 420.0 -> (420.0 - m) / 30.0 * 1.5
            else -> (m - 510.0) / 30.0 * 0.75
        }.coerceIn(-1.0, 8.0)
    }

    private fun activityOffset(steps: Double?, strain: Double?): Double? {
        val stepsOffset = steps?.let { ((7000.0 - it) / 2000.0 * 2.0).coerceIn(-5.0, 6.0) }
        val strainOffset = strain?.let { ((8.0 - it) * 0.4).coerceIn(-3.0, 3.0) }
        return when {
            stepsOffset != null && strainOffset != null -> 0.6 * stepsOffset + 0.4 * strainOffset
            else -> stepsOffset ?: strainOffset
        }
    }
}

package com.esusic.recoverytracker.scoring

import java.time.LocalDate

data class DailyVitals(
    val date: LocalDate,
    val hrvMillis: Double? = null,
    val restingHr: Double? = null,
    val sleepMinutes: Double? = null,
    val steps: Double? = null,
    val strain: Double? = null
)

object VitalAgeSeries {
    const val WINDOW_DAYS = 14L
    const val MIN_WINDOW_DAYS = 3

    fun build(days: List<DailyVitals>, birthYear: Int): List<Pair<LocalDate, VitalAgeResult>> {
        val byDate = days.associateBy { it.date }
        return days.map { it.date }.sorted().mapNotNull { date ->
            val window = (0 until WINDOW_DAYS).mapNotNull { byDate[date.minusDays(it)] }
            if (window.size < MIN_WINDOW_DAYS) return@mapNotNull null
            val inputs = VitalAgeInputs(
                avgHrvMillis = window.mapNotNull { it.hrvMillis }.averageOrNull(),
                avgRestingHr = window.mapNotNull { it.restingHr }.averageOrNull(),
                avgSleepMinutes = window.mapNotNull { it.sleepMinutes }.averageOrNull(),
                avgSteps = window.mapNotNull { it.steps }.averageOrNull(),
                avgStrain = window.mapNotNull { it.strain }.averageOrNull()
            )
            VitalAgeCalculator.calculate(date.year - birthYear, inputs)?.let { date to it }
        }
    }

    private fun List<Double>.averageOrNull(): Double? = if (isEmpty()) null else average()
}

                    hrvMillis = aggregate.avgHrvMillis,
                    restingHr = aggregate.restingHeartRate?.toDouble(),
                    sleepMinutes = aggregate.sleepDurationMinutes?.toDouble(),
                    steps = aggregate.steps.takeIf { it > 0 }?.toDouble(),
                    strain = strainByDate[aggregate.date]
                )
            }

            val series = VitalAgeSeries.build(days, birthYear)
            val latest = series.lastOrNull()
            val weekBefore = latest?.let { (date, _) -> series.find { it.first == date.minusDays(7) } }
            val weeklyDelta = if (latest != null && weekBefore != null) {
                val diff = latest.second.vitalAge - weekBefore.second.vitalAge
                MetricDelta(
                    diff,
                    when {
                        abs(diff) < 0.1 -> DeltaKind.NEUTRAL
                        diff < 0 -> DeltaKind.BETTER
                        else -> DeltaKind.WORSE
                    }
                )
            } else {
                null
            }

            _uiState.value = VitalAgeUiState(
                isLoading = false,
                chronologicalAge = LocalDate.now(clock).year - birthYear,
                result = latest?.second,
                weeklyDelta = weeklyDelta,
                history = series.map { (date, result) -> ChartPoint(date, result.vitalAge.toFloat()) },
                daysUntilReady = if (latest == null) {
                    (VitalAgeSeries.MIN_WINDOW_DAYS - days.size).coerceAtLeast(1)
                } else {
                    null
                }
            )
        }
    }

    companion object {
        const val HISTORY_DAYS = 44
    }
}
```

@Composable
fun VitalAgeScreen(viewModel: VitalAgeViewModel) {
    val uiState by viewModel.uiState.collectAsState()

    Column(
        modifier = Modifier
            .fillMaxSize()
            .verticalScroll(rememberScrollState())
            .padding(horizontal = 20.dp, vertical = 24.dp),
        verticalArrangement = Arrangement.spacedBy(20.dp)
    ) {
        Eyebrow("Vital age")

        val result = uiState.result
        when {
            uiState.isLoading -> Unit
            result == null -> CalibratingCard(uiState.daysUntilReady)
            else -> {
                HeroSection(result, uiState)
                AgingPaceCard(uiState)
                BreakdownCard(result)
                if (uiState.history.size >= 2) HistoryCard(uiState.history)
            }
        }
    }

@Composable
private fun HeroSection(result: VitalAgeResult, uiState: VitalAgeUiState) {
    Column(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        Row(verticalAlignment = Alignment.Bottom, horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            Text(
                "%.1f".format(result.vitalAge),
                fontSize = 64.sp,
                fontWeight = FontWeight.Bold,
                color = AppColors.VitalAge
            )
            Text(
                "yrs",
                style = MaterialTheme.typography.titleMedium,
                color = AppColors.TextMuted,
                modifier = Modifier.padding(bottom = 12.dp)
            )
        }
        val diff = result.vitalAge - result.chronologicalAge
        val (text, color) = when {
            abs(diff) < 0.5 -> "Matches your age (${result.chronologicalAge})" to AppColors.TextSecondary

            else -> "%.1f yrs older than your age (%d)".format(diff, result.chronologicalAge) to AppColors.Danger
        }
        Text(text, style = MaterialTheme.typography.titleSmall, color = color, fontWeight = FontWeight.SemiBold)
        DeltaChip(delta = uiState.weeklyDelta, unit = "yr", decimals = 1, suffix = "vs last week")
    }
}

                    component.offsetYears <= -0.05 -> AppColors.Recovery
                    component.offsetYears >= 0.05 -> AppColors.Danger
                    else -> AppColors.TextSecondary
                }
            )
        }

@Composable
private fun DivergingBar(offsetYears: Double?) {
    val fraction = ((offsetYears ?: 0.0) / 10.0).toFloat().coerceIn(-1f, 1f)
    Row(modifier = Modifier.fillMaxWidth().height(6.dp)) {
        Box(
            modifier = Modifier.weight(1f).fillMaxHeight()
                .clip(RoundedCornerShape(topStart = 3.dp, bottomStart = 3.dp))
                .background(MaterialTheme.colorScheme.surfaceVariant),
            contentAlignment = Alignment.CenterEnd
        ) {
            if (fraction < 0f) {
                Box(
                    modifier = Modifier.fillMaxWidth(-fraction).fillMaxHeight()
                        .background(AppColors.Recovery)
                )
            }
        }
        Spacer(modifier = Modifier.width(2.dp))
        Box(
            modifier = Modifier.weight(1f).fillMaxHeight()
                .clip(RoundedCornerShape(topEnd = 3.dp, bottomEnd = 3.dp))
                .background(MaterialTheme.colorScheme.surfaceVariant),
            contentAlignment = Alignment.CenterStart
        ) {
            if (fraction > 0f) {
                Box(
                    modifier = Modifier.fillMaxWidth(fraction).fillMaxHeight()
                        .background(AppColors.Danger)
                )
            }
        }
    }
}
